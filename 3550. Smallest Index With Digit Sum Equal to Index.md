  # 3550. Smallest Index With Digit Sum Equal to Index 
        
## Solution

```
    class Solution:
    def smallestIndex(self, nums: List[int]) -> int:
        n = len(nums)

        for i in range(n):
            curr = nums[i]

            curr = str(curr)
            for s in curr:
                ans += int(s)
            
            if ans == i:
                return i

            ans = 0
            print(curr, ans)
        return -1
```
